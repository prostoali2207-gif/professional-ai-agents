```python
import hashlib, json, re, runpy, socket, sys
from pathlib import Path

root = Path.cwd()
stage = root / 'architect/evaluation/qualification-reliability-engineer/stage-b'
def blocked(*args, **kwargs):
    raise RuntimeError('Network prohibited during B1 local validation')
socket.socket = blocked
socket.create_connection = blocked
sys.path.insert(0, str(stage))
from validate_calibration_manifest import REQUIRED_MECHANISMS, REQUIRED_LEVELS, REQUIRED_RUBRIC

p = json.loads((stage/'calibration-pack-v0.1.json').read_bytes())
m = json.loads((stage/'calibration-manifest-v0.1.json').read_bytes())
ref = (stage/'calibration-reference-v0.1.md').read_text(encoding='utf-8')
assert hashlib.sha256((stage/'calibration-pack-v0.1.json').read_bytes()).hexdigest() == m['pack']['sha256']
assert len(p['cases']) == m['pack']['case_count'] == 12
assert len({c['case_id'] for c in p['cases']}) == 12
assert {c['mechanism'] for c in p['cases']} == set(m['mechanisms']) == REQUIRED_MECHANISMS
assert {r['dimension'] for r in p['rubric']} == set(m['rubric_dimensions']) == REQUIRED_RUBRIC
assert {d for c in p['cases'] for d in c['judgment_dimensions']} == REQUIRED_RUBRIC
assert set(p['policy']['p0_rules']) == {f'P0-{n:02}' for n in range(1,11)}
ops = {'eq':lambda x:x[0]==x[1], 'sum':sum, 'subtract':lambda x:x[0]-x[1],
       'multiply':lambda x:x[0]*x[1], 'lte':lambda x:x[0]<=x[1]}
assertions = 0
ids = set()
for c in p['cases']:
    for field in ['facts_observables','unknowns_misleading_cues','decision_relevant_variables',
                  'expected_decision_properties','unacceptable_shortcuts','applicable_p0_triggers',
                  'deterministic_assertions','judgment_dimensions','competencies','legitimate_disagreement_boundary']:
        assert c[field], (c['case_id'],field)
    assert len(c['reference_exemplars']) == 4
    for r in c['reference_exemplars']:
        assert r['reference_id'] not in ids
        ids.add(r['reference_id'])
        assert set(r) == {'reference_id','response'}
    for a in c['deterministic_assertions']:
        assert ops[a['operation']](a['inputs']) == a['expected'], a['assertion_id']
        assertions += 1
rows = re.findall(r'^\| (B1-\d+) \| (R\d+) \| ([A-Z_]+) \| ([^\n]+) \| ([^\n]+) \|$',ref,re.M)
assert len(rows)==48
assert {r[1] for r in rows} == ids
assert {r[2] for r in rows} == REQUIRED_LEVELS
unsafe_p0 = set()
for cid,rid,level,reason,triggers in rows:
    c = next(c for c in p['cases'] if c['case_id']==cid)
    assert rid in {r['reference_id'] for r in c['reference_exemplars']}
    referenced = set(re.findall(r'P0-\d+',triggers))
    assert referenced <= {a['p0_id'] for a in c['applicable_p0_triggers']}
    if level=='UNSAFE_NAIVE':
        assert referenced
        unsafe_p0.update(referenced)
assert unsafe_p0 == set(p['policy']['p0_rules'])
print(json.dumps({'status':'B1_CONTENT_CONSISTENCY_PASS','cases':12,'references':48,
 'mechanisms':12,'rubric_dimensions':14,'levels':5,'unsafe_exemplar_p0_coverage':10,
 'synthetic_arithmetic_equality_assertions':assertions,'sha256':m['pack']['sha256'],
 'network':'socket operations disabled','scope':'authored data consistency only; no B2 or candidate execution'}))
```
